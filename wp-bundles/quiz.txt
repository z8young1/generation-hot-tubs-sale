    var pad = function (n) { return (n < 10 ? "0" : "") + n; };

      if (now >= end) {

        if (label) label.textContent = "Doors are open";
        return;

      var s = Math.floor((start - now) / 1000);
      var d = Math.floor(s / 86400); s -= d * 86400;
      var h = Math.floor(s / 3600); s -= h * 3600;
      var m = Math.floor(s / 60);
      timer.innerHTML =
        "<span><b>" + d + "</b>days</span>" +
        "<span><b>" + pad(h) + "</b>hrs</span>" +
        "<span><b>" + pad(m) + "</b>min</span>";

      if (bad) { f.setAttribute("data-bad", "1"); input.setAttribute("aria-invalid", "true"); ok = false; }

(function () {
  var order = ["people", "purpose", "space", "features", "timing"];
  var answers = {};
  var step = 0; // 0..4 questions, 5 = contact
  var qs = {};
  order.forEach(function (k) { qs[k] = document.querySelector('[data-q="' + k + '"]'); });
  var intro = document.getElementById("intro");
  var contact = document.getElementById("contact");
  var back = document.getElementById("back");
  var hint = document.getElementById("hint");
  var bar = document.getElementById("bar");
  var pct = document.getElementById("pct");
  var label = document.getElementById("step-label");
  var tMatch = document.getElementById("t-match");
  var tName = document.getElementById("t-name");

  window.GHTQuizAnswers = function () { return answers; };

  function render() {
    order.forEach(function (k, i) { qs[k].hidden = (i !== step); });
    intro.hidden = step !== 0;
    contact.hidden = step !== 5;
    back.hidden = step === 0;
    hint.hidden = step === 5;
    var done = Math.min(step, 5);
    bar.style.width = (done / 5 * 100) + "%";
    pct.textContent = Math.round(done / 5 * 100) + "%";
    label.textContent = step < 5 ? "Question " + (step + 1) + " of 5" : "Last step";
    if (step >= 3 && window.GHTLead) tMatch.textContent = window.GHTLead.match(answers).name;
    if (step === 5) { var f = contact.querySelector("input"); if (f) f.focus(); }
    window.scrollTo({ top: 0, behavior: "smooth" });
  }

  document.querySelectorAll(".opt").forEach(function (btn) {
    btn.addEventListener("click", function () {
      var key = btn.closest("[data-q]").getAttribute("data-q");
      answers[key] = btn.getAttribute("data-v");
      btn.parentElement.querySelectorAll(".opt").forEach(function (b) { b.setAttribute("aria-pressed", b === btn ? "true" : "false"); });
      setTimeout(function () { step = Math.min(step + 1, 5); render(); }, 160);
    });
  });
  back.addEventListener("click", function () { step = Math.max(step - 1, 0); render(); });

  // Letter keys A to D answer the visible question
  document.addEventListener("keydown", function (e) {
    if (step > 4 || e.metaKey || e.ctrlKey || e.altKey) return;
    var i = "abcd".indexOf(e.key.toLowerCase());
    if (i < 0) return;
    var opt = qs[order[step]].querySelectorAll(".opt")[i];
    if (opt) opt.click();
  });

  document.getElementById("first_name").addEventListener("input", function (e) {
    tName.textContent = e.target.value.trim() || "Your name here";
  });

  render();
})();
